	\section{La soluzione}

	\begin{frame}{La soluzione}
		\vfill
		{\centering \huge{\textbf{\textcolor{rossoPantano}{Architettura bottom-up}}} \par}
		\vspace{0.5em}
		Sviluppo incrementale articolato in quattro layer: \vspace{.5em}
		\begin{enumerate}
			\item \textbf{Modellazione dei dati:} entità di dominio (\texttt{ClientModel}, \texttt{SessionModel}) ed enumeratori crittografici. \vspace{.5em}
			\item \textbf{Layer di accesso ai dati:} \textit{repository} su \textbf{Redis} (sessioni) e \textbf{MongoDB} (configurazioni). \vspace{.5em}
			\item \textbf{Layer di logica applicativa:} servizi di inizializzazione, elaborazione risposta e validazione crittografica. \vspace{.5em}
			\item \textbf{Layer di presentazione:} \textit{route} Express e interfacce REST.
		\end{enumerate}
		\vfill
	\end{frame}

	\begin{frame}{La soluzione}
		\vfill
		{\centering \huge{\textbf{\textcolor{rossoPantano}{Sicurezza by design}}} \par}
		\vspace{0.5em}
		Come garantire la solidità di un sistema crittografico? \vspace{.5em}
		\begin{itemize}
			\item \textbf{Verifica statica:} type checking rigoroso (\texttt{alwaysStrict: true}) per vincolare staticamente gli algoritmi crittografici supportati. \vspace{.5em}
			\item \textbf{Misure difensive:} validazione preventiva e isolamento \textit{URL-safe} degli input per evitare manipolazioni e \textit{injection}. \vspace{.5em}
			\item \textbf{Audit automatico:} controllo continuo delle vulnerabilità nelle dipendenze eseguito ad ogni riavvio (\texttt{yarn audit}).
		\end{itemize}
		\vfill
	\end{frame}

	\begin{frame}{La soluzione}
		\vfill
		{\centering \huge{\textbf{\textcolor{rossoPantano}{Validazione Crittografica}}} \par}
		\vspace{0.5em}
		\begin{block}{OpenID Federation (Trust Chain)}
			Navigazione ricorsiva tramite visita in profondità (DFS) con \textit{cycle detection} per validare l'intera catena di fiducia.
		\end{block}

		\begin{alertblock}{SD-JWT (Selective Disclosure)}
			Parsifica del \textit{token}, controllo della firma dell'\textit{Issuer}, validazione degli \textit{hash} (disclosures) e verifica del \textit{Key Binding}.
		\end{alertblock}

		\begin{exampleblock}{Query DCQL}
			Parsing strutturato e validazione semantica avanzata delle richieste formali espresse in \textit{Digital Credentials Query Language}.
		\end{exampleblock}
		\vfill
	\end{frame}

	\begin{frame}{La soluzione}
		\vfill
		{\centering \huge{\textbf{\textcolor{rossoPantano}{Affidabilità}}} \par}
		\vspace{0.5em}
		Come evitare inconsistenze di stato per le sessioni utente? \vspace{.5em}
		\begin{itemize}
			\item \textbf{Stato su Redis:} ogni sessione ha un \textit{TTL} automatico e le chiavi asimmetriche (Forward Secrecy) vengono distrutte atomicamente post-utilizzo. \vspace{.5em}
			\item \textbf{Scalabilità orizzontale:} assenza di stato in memoria, i server possono essere replicati nativamente dietro a un load balancer. \vspace{.5em}
			\item \textbf{Multi-tenancy nativa:} isolamento per dominio garantito sia a livello applicativo che sui database tramite indici composti.
		\end{itemize}
		\vfill
	\end{frame}
	
	\begin{frame}{La soluzione}
		\vfill
		{\centering \huge{\textbf{\textcolor{rossoPantano}{Validazione Formale}}} \par}
		\vspace{0.5em}
		\begin{itemize}
			\item \textbf{Conformance Test:} superamento dell'intera suite di certificazione ufficiale fornita dalla \textbf{OpenID Foundation}. \vspace{.5em}
			\item \textbf{Profilo Verificato:} superamento completo del profilo \textit{OID4VP Verifier}. \vspace{.5em}
			\item \textbf{Obiettivo:} garanzia di interoperabilità assoluta con qualsiasi \textit{Wallet} certificato europeo presente sul mercato.
		\end{itemize}
		\vfill
	\end{frame}
