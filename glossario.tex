% --- Voci di glossario ---
\newglossaryentry{apig}
{
    name=API,
    text=Application Program Interface,
    sort=api,
    description={in informatica con il termine \emph{Application Programming Interface API} si indica ogni insieme di procedure disponibili al programmatore, di solito raggruppate a formare un set di strumenti specifici per il completamento di un determinato compito all'interno di un certo programma}
}

\newglossaryentry{umlg}
{
    name=UML,
    text=UML,
    sort=uml,
    description={in ingegneria del software \emph{UML, Unified Modeling Language} è un linguaggio di modellazione e specifica basato sul paradigma object-oriented. Gran parte della documentazione tecnica lo utilizza per descrivere architetture e soluzioni software in modo sintetico e universalmente comprensibile}
}

\newglossaryentry{dcqlg}
{
    name=DCQL,
    text=DCQL,
    sort=dcql,
    description={acronimo di \emph{Digital Credentials Query Language}, è un linguaggio di interrogazione definito nell'ambito della specifica OpenID for Verifiable Presentations (OID4VP) che consente al Verifier di esprimere in modo strutturato e granulare i requisiti relativi alle credenziali verificabili richieste all'utente, specificando tipologie di credenziale, attributi da rivelare e autorità emittenti fidate}
}

\newglossaryentry{sdjwtg}
{
    name=SD-JWT,
    text=SD-JWT,
    sort=sdjwt,
    description={acronimo di \emph{Selective Disclosure JSON Web Token}, è un formato di credenziale verificabile basato su JWT che consente al titolare di rivelare selettivamente solo un sottoinsieme degli attributi contenuti nella credenziale}
}

\newglossaryentry{forwardsecrecyg}
{
    name=Forward Secrecy,
    text=Forward Secrecy,
    sort=forward secrecy,
    description={proprietà di un protocollo crittografico per cui la compromissione delle chiavi di lungo periodo non compromette la riservatezza delle sessioni di comunicazione precedentemente instaurate. Viene ottenuta mediante l'impiego di chiavi crittografiche effimere generate ad hoc per ciascuna sessione e distrutte al termine dell'utilizzo}
}

\newglossaryentry{scimg}
{
    name=SCIM,
    text=SCIM,
    sort=scim,
    description={acronimo di \emph{System for Cross-domain Identity Management}, è un protocollo standardizzato per l'automazione dello scambio di informazioni di identità tra domini, che definisce schemi e API REST per la gestione del ciclo di vita delle risorse di identità in ambienti multi-tenant}
}

\newglossaryentry{multitenancyg}
{
    name=Multi-tenancy,
    text=multi-tenancy,
    sort=multi-tenancy,
    description={architettura software in cui una singola istanza di un'applicazione serve molteplici organizzazioni o clienti (\emph{tenant}), garantendo l'isolamento logico dei dati e delle configurazioni tra ciascun tenant pur condividendo la medesima infrastruttura}
}

\newglossaryentry{oid4vpg}
{
    name=OID4VP,
    text=OID4VP,
    sort=oid4vp,
    description={acronimo di \emph{OpenID for Verifiable Presentations}, è una specifica del framework OpenID che definisce il protocollo di interazione tra un Verifier (Relying Party) e un Digital Wallet per la richiesta, la trasmissione e la verifica di presentazioni di credenziali verificabili}
}

\newglossaryentry{jweg}
{
    name=JWE,
    text=JWE,
    sort=jwe,
    description={acronimo di \emph{JSON Web Encryption}, è uno standard per la cifratura di contenuti in formato compatto e sicuro basato su JSON, utilizzato per proteggere la riservatezza dei dati trasmessi tra le parti di un protocollo crittografico}
}

\newglossaryentry{jwsg}
{
    name=JWS,
    text=JWS,
    sort=jws,
    description={acronimo di \emph{JSON Web Signature}, è uno standard per la firma digitale di contenuti in formato JSON, utilizzato per garantire l'integrità e l'autenticità dei dati trasmessi}
}

\newglossaryentry{jwtg}
{
    name=JWT,
    text=JWT,
    sort=jwt,
    description={acronimo di \emph{JSON Web Token}, è un formato compatto e autonomo per la rappresentazione sicura di attestazioni (\emph{claims}) tra due parti, tipicamente utilizzato per l'autenticazione e lo scambio di informazioni}
}

\newglossaryentry{jwkg}
{
    name=JWK,
    text=JWK,
    sort=jwk,
    description={acronimo di \emph{JSON Web Key}, è un formato standard per la rappresentazione di chiavi crittografiche in formato JSON}
}

\newglossaryentry{trustchaing}
{
    name=Trust Chain,
    text=Trust Chain,
    sort=trust chain,
    description={nel contesto di OpenID Federation 1.0, è una catena gerarchica di attestazioni digitali firmate che collega un'entità foglia (ad esempio un Issuer o un Verifier) a un'ancora di fiducia (\emph{Trust Anchor}) riconosciuta, permettendo di dedurre e verificare la legittimità dell'entità senza conoscenza diretta preventiva}
}

\newglossaryentry{vptokeng}
{
    name=VP Token,
    text=VP Token,
    sort=vp token,
    description={nel protocollo OID4VP, è il token che incapsula una o più presentazioni verificabili (\emph{Verifiable Presentations}) trasmesse dal Wallet al Verifier in risposta a una richiesta di presentazione}
}

\newglossaryentry{ttlg}
{
    name=TTL,
    text=TTL,
    sort=ttl,
    description={acronimo di \emph{Time-To-Live}, indica il tempo massimo di validità o permanenza di un dato in un sistema di memorizzazione. In Redis, il TTL definisce il tempo trascorso il quale una chiave viene automaticamente rimossa dal datastore}
}

\newglossaryentry{jarg}
{
    name=JAR,
    text=JAR,
    sort=jar,
    description={acronimo di \emph{JWT-Secured Authorization Request}, è una specifica che richiede che la richiesta di autorizzazione OAuth 2.0 sia incapsulata in un JWT firmato digitalmente, garantendo l'integrità e l'autenticità dei parametri della richiesta}
}

\newglossaryentry{dtog}
{
    name=DTO,
    text=DTO,
    sort=dto,
    description={acronimo di \emph{Data Transfer Object}, è un pattern architetturale che definisce un oggetto il cui unico scopo è il trasporto di dati tra i livelli di un'applicazione, privo di logica di business}
}

\newglossaryentry{kbjwtg}
{
    name=KB-JWT,
    text=KB-JWT,
    sort=kb-jwt,
    description={acronimo di \emph{Key Binding JWT}, è un token JWT firmato dal dispositivo del titolare della credenziale (\emph{Holder}) con la chiave privata associata alla credenziale SD-JWT. La sua verifica attesta il possesso effettivo della credenziale da parte del presentatore (\emph{Proof of Possession})}
}

\newglossaryentry{iamg}
{
    name=IAM,
    text=IAM,
    sort=iam,
    description={acronimo di \emph{Identity and Access Management}, indica l'insieme di processi, tecnologie e politiche volti a gestire le identità digitali e a regolamentare l'accesso alle risorse di un sistema informatico}
}

\newglossaryentry{cicdg}
{
    name=CI/CD,
    text=CI/CD,
    sort=cicd,
    description={acronimo di \emph{Continuous Integration / Continuous Delivery}, indica l'insieme di pratiche di ingegneria del software che automatizzano l'integrazione frequente del codice, l'esecuzione dei test e il rilascio del software in ambienti di produzione}
}


\newglossaryentry{crudg}
{
    name=CRUD,
    text=CRUD,
    sort=crud,
    description={acronimo di \emph{Create, Read, Update, Delete}, indica le quattro operazioni fondamentali di gestione dei dati persistenti in un sistema informatico}
}

\newglossaryentry{replayattackg}
{
    name=Replay Attack,
    text=Replay Attack,
    sort=replay attack,
    description={tipologia di attacco informatico in cui un avversario intercetta e ritrasmette un messaggio o una credenziale legittima precedentemente catturata, al fine di ottenere accesso non autorizzato o di eseguire operazioni fraudolente}
}

\newglossaryentry{oauthg}
{
    name=OAuth 2.0,
    text=OAuth 2.0,
    sort=oauth,
    description={framework di autorizzazione standardizzato che consente a un'applicazione di terze parti di ottenere un accesso limitato a un servizio HTTP, sia per conto di un proprietario della risorsa, sia per conto proprio. OID4VP si basa sulle estensioni di questo framework}
}

\newglossaryentry{relyingpartyg}
{
    name=Relying Party,
    text=Relying Party,
    sort=relying party,
    description={nell'ambito dell'identità digitale, è l'entità o il servizio che si affida alle credenziali presentate da un utente e verificate tramite un'infrastruttura fidata per concedere l'accesso a una risorsa o fornire un servizio}
}

\newglossaryentry{verifierg}
{
    name=Verifier,
    text=Verifier,
    sort=verifier,
    description={sinonimo di Relying Party nell'ecosistema OID4VP. Identifica il ruolo dell'entità che richiede una presentazione verificabile (\emph{Verifiable Presentation}), ne valida l'autenticità e l'integrità crittografica prima di accettare le credenziali}
}

\newglossaryentry{digitalwalletg}
{
    name=Digital Wallet,
    text=Digital Wallet,
    sort=digital wallet,
    description={applicazione software o servizio che consente a un utente di conservare, gestire e presentare in modo sicuro le proprie credenziali digitali (es. EUDI Wallet), agendo come tramite tra l'utente, le autorità emittenti (Issuer) e i fornitori di servizi (Verifier)}
}
